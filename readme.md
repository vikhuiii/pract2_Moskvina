
git init - иницализация
git status - статус репозитория
git add file или . - добавить файл 
git commit -m "c" - зафиксировать изменения
git log - история коммитов
git branch name- создать ветку
git branch - список веток
git switch name - перейти на ветку(новое)
git checkout name - перейти на ветку(старое)
git merge name - слить указанную ветку в текущую
git branch -d name - удалить ветку
git remote add origin s - привязка к удаленному репозиторию
git remote -v - проверка к какому репозиторию привязан
git push -u origin master - первый пуш
git push - последующие пуши
git branch -r - список удаленных веток
git fetch origin/name- загрузить изменения из удаленного репозитория в локальную копию
git diff name origin/name - различия в содержимом файлов
git pull - напрямую получить в локальную ветку файлы из удаленного репозитория
